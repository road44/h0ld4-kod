    id,
    nazwa,
    wystepowanie
FROM
    ryby
WHERE
    styl_zycia = 2;

-- kw.2
SELECT
    akwen,
    wojewodztwo
FROM
    lowisko
WHERE
    rodzaj = 2
    OR rodzaj = 3;

-- kw.3
SELECT
    nazwa
FROM
    ryby r
    INNER JOIN okres_ochronny o ON o.ryby_id = r.id
WHERE
    o.wymiar_ochronny = 0;
